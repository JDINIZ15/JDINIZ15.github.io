create database pokedex;

use pokedex;



create table tipagem(
 tipo1 varchar (50),
 tipo2 varchar (50),
 idTipagem int primary key
);

create table Ovos(
 grupo1 varchar (50),
 grupo2 varchar (50),
 idOvos int primary key
);

create table regiao (
	geracao varchar (50),
	nome varchar (50),
	localizacao varchar (50),
	idRegiao int primary key
);

create table moveset(
	ataque1 varchar (50),
	ataque2 varchar (50),
	ataque3 varchar (50),
	ataque4 varchar (50),
	idMoveset int primary key
);


create table Pokemon (
    idpokemon int, 
    idTipagem int,
    idOvos int,
    nome varchar (50),
    habilidade varchar (50), 
    nature varchar (50),
    primary key (idpokemon),
    datacap date,
    level int,
    foreign key (idTipagem) references tipagem (idTipagem),
    foreign key (idOvos) references ovos(idOvos)

);


create table pokemove(
    idpokemon int,
    idMoveset int,
    idPokemove int,
    PRIMARY key (idPokemove),
    foreign key (idMoveset) references moveset(idMoveset),
    foreign key (idpokemon) references pokemon(idpokemon)

);

create table pokereg(
    idpokemon int,
    idRegiao int,
    idPokereg int,
    PRIMARY key (idPokereg),
    foreign key (idpokemon) references pokemon(idpokemon),
    foreign key (idRegiao) references Regiao(idRegiao)
);
