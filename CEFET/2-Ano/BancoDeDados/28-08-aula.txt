/* Tabelas */
/* Chris & Samara =) */ 
create table produto(
	id int auto_increment primary key,
	nome varchar(45),
	descricao varchar(200),
	preco float,
	qtdestoque int
);

create table item(
	id int AUTO_INCREMENT primary key,
	idvenda int,
	idproduto int,
	qtd int
);

create table venda(
	id int auto_increment primary key,
	data date,
	hora time,
	idCliente int
);

create Table Cliente (     
	id int NOT NULL AUTO_INCREMENT primary key,
 	nome VARCHAR(45),
	uf char(2),
	rendamensal float
);


/* Alter Tables */


alter table item add foreign key(idproduto) references produto(id);
alter table item add foreign key(idvenda) references venda(id);
alter table venda add foreign key(idCliente) references Cliente(id);
