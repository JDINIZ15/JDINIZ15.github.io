CREATE DATABASE aluguel CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;
USE aluguel;

CREATE TABLE Inquilino (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    Nome VARCHAR(100) NOT NULL,
    CPF VARCHAR(11) UNIQUE NOT NULL,
    Telefone VARCHAR(15),
    Endereco VARCHAR(255)
);

CREATE TABLE Imovel (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    Endereco VARCHAR(255) NOT NULL,
    Tipo VARCHAR(50) NOT NULL,
    Valor_Aluguel DECIMAL(10, 2) NOT NULL,
    Situacao ENUM('disponivel', 'alugado', 'manutencao') DEFAULT 'disponivel'
);

CREATE TABLE Aluguel (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    Data_Inicio DATE NOT NULL,
    Data_Termino DATE,
    Valor_Aluguel DECIMAL(10, 2) NOT NULL,
    Status_Contrato ENUM('ativo', 'encerrado') DEFAULT 'ativo',
    Inquilino_ID INT NOT NULL,
    Imovel_ID INT NOT NULL,
    FOREIGN KEY (Inquilino_ID) REFERENCES Inquilino(ID),
    FOREIGN KEY (Imovel_ID) REFERENCES Imovel(ID)
);

# Criar tabelas com chaves primarias e estrangeiras utilizando o script

create table Categoria (

	idCategoria int not null,
	nome varchar(30) not null,
    
	primary key (idCategoria)
);

create table Produto (

	idProduto int not null,
 	nome varchar(45) not null,
	preco decimal(10,2),
	idcategoria int,
    
	primary key (idProduto),
	foreign key (idCategoria) references Categoria(idCategoria)
);

create table Cliente (

	idCliente int not null,
 	nome varchar(45) not null,
    data_nasc date not null,
    logradouro varchar(45),
    numero int,
    bairro varchar(45),
    cidade varchar(45),
    telefone varchar(20),
    
	primary key (idCliente)
);

create table Pedido (

	idPedido int not null,
	data_emissao date not null,
    idcliente int not null,
    
	primary key (idPedido),
	foreign key (idCliente) references Cliente(idCliente)
);

# Implementacao da relacao MxN com uso de ID unico na tabela intermediaria
# Chave primaria com autoincremento para IDs (chaves artificiais)

create table ItensPedido (

	id int AUTO_INCREMENT,
	idPedido int not null,
    idProduto int not null,
    qtde int not null,
    
	primary key (id),
	foreign key (idPedido) references Pedido(idPedido),
	foreign key (idProduto) references Produto(idProduto)
);

# Join com 3 tabelas

# Exemplo
# Listar para cada pedido o numero do pedido, a data de emissão, 
# o código, nome, preco e qtde dos itens do pedido.

select Pedido.idPedido, 
	   Pedido.data_emissao,
       Produto.idProduto,
       Produto.nome,
       Produto.preco,
       ItensPedido.qtde

	from ItensPedido

    join Pedido
    on ItensPedido.idPedido = Pedido.idPedido

    join Produto
    on ItensPedido.idProduto = Produto.idProduto
    
    order by ItensPedido.idPedido, ItensPedido.idProduto;




# INNER JOIN
# Método de junção mais conhecido, retorna os registros que são comuns às duas tabelas

# Exemplo
# Listas as categorias e os produtos das categorias
# Obs: categorias que não tenham produtos associados não são incluídas no resultado;

select c.nome, p.nome, p.preco
	from categoria c
    inner join produto p
    on p.idcategoria = c.idCategoria
    order by c.nome, p.nome;

# O inner join é aplicado como default para o join
# sem usar a palavra-chave inner, o resultado é o mesmo

select c.nome, p.nome, p.preco
	from categoria c
    join produto p
    on p.idcategoria = c.idCategoria
    order by c.nome, p.nome;


# LEFT JOIN
# Tem como resultado todos os registros que estão na tabela A (mesmo que não estejam na tabela B)
# e os registros da tabela B que são comuns à tabela A

# Exemplo
# Listas todas as categorias (mesmo que não tenha produtos associados) e os produtos das categorias
# A tabela considerada LEFT é a que é citada PRIMEIRO no script (neste exemplo, a tabela categoria)

select c.nome, p.nome, p.preco
	from categoria c
    left join produto p
    on p.idcategoria = c.idCategoria
    order by c.nome, p.nome;


# RIGHT JOIN
# Tem como resultado todos os registros que estão na tabela B (mesmo que não estejam na tabela A) 
# e os registros da tabela A que são comuns à tabela B.

# Exemplo: o mesmo se aplica o exemplo anterior com as tabelas citadas com ordem invertida
# O mesmo se aplica ao RIGHT JOIN, porem a tabela total será a SEGUNDA tabela citada no script

select c.nome, p.nome, p.preco
	from produto p
    right join categoria c
    on p.idcategoria = c.idCategoria
    order by c.nome, p.nome;

# Exemplo:
# Exibir todos os produto, com seus preços e descrição da categoria (se houver associação)

select p.nome, p.preco, c.nome
	from categoria c
    right join produto p
    on p.idcategoria = c.idCategoria
    order by p.nome, c.nome;


# OUTER JOIN
# O OUTER JOIN (também conhecido por Full Outer Join ou Full Join), 
# tem como resultado todos os registros que estão na tabela A e todos os registros da tabela B.

# ** O MySQL não suporte o OUTER JOIN
# ** Neste caso, utiliza-se a clausula UNION para juntar um LEFT JOIN com um RIGHT JOIN

# UNION
# Combina o resultado de execução das duas queries e então executa um SELECT DISTINCT 
# a fim de eliminar as linhas duplicadas. Este processo é executado mesmo que não hajam registros duplicados.


# Exemplo: 
# Exibir todos os produtos e categorias cadastrados

select categoria.nome, produto.nome, produto.preco

	from categoria
    left join produto
    on produto.idcategoria = categoria.idCategoria
    
    union
    
	select categoria.nome, produto.nome, produto.preco
	from categoria
    right join produto
    on produto.idcategoria = categoria.idCategoria;
