/*7. Relacionamento: medico -> paciente*/

 create table medico(
	idmed int,
	nome varchar(35) not null,
	primary key(idmed)
);

 create table plano(
	idpla int,
	nomeplan varchar(20),
	primary key(idpla)
);
 create table paciente(
	idpac int,
	nome varchar(30),
	idpla int,
	primary key(idpac),
;	foreign key (idpla) references plano(idpla)
)

Create table consulta(
	idcons int primary key auto_increment,
	idmed int,
	idpac int,
	datcon date,
	prescricao varchar(3000),
	foreign key (idmed) references medico (idmed),
	foreign key (idpac) references paciente (idpac)
)
