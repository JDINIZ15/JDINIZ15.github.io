/*Letra B*/
Create table cargo (
codigo bigint primary key,
nome varchar(50)
);


Create table departamento(
codigo bigint primary key,
nome varchar(50),
funcionario_gerente bigint
);



Create table funcionario(
codigo bigint primary key,
funcionario_gerente bigint,
nome varchar(50),
data_contratacao date,
salario bigint,
cod_cargo bigint,
departamento_alocado bigint
);

alter table funcionario add foreign key (cod_cargo) references cargo(codigo);


alter table funcionario add foreign key (departamento_alocado) references departamento(codigo);


alter table departamento add foreign key (funcionario_gerente) references funcionario(codigo);
